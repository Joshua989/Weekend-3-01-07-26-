35const IssueStore = {
36 /** Return every saved issue (newest first). */
37 getAll() {
38 try {
39 const raw = localStorage.getItem(IRO.STORAGE_KEY);
40 const list = raw ? JSON.parse(raw) : [];
41 return list.sort((a, b) => b.createdAt.localeCompare(a.createdAt));
42 } catch (e) {
43 return [];
44 }
45 },
46
47 /** Persist the full list. */
48 saveAll(list) {
49 localStorage.setItem(IRO.STORAGE_KEY, JSON.stringify(list));
50 },
51
52 /** Create a unique, human-friendly ticket ID such as IRO-26-4F7K2. */
53 generateId() {
54 const chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"; // no 0/O/1/I to avoid confusion
55 const year = String(new Date().getFullYear()).slice(-2);
56 let id;
57 do {
58 let code = "";
59 for (let i = 0; i < 5; i++) code += chars[Math.floor(Math.random() * chars.length)];
60 id = `IRO-${year}-${code}`;
61 } while (this.findById(id));
62 return id;
63 },
64
65 /** Add a new issue and return it. */
66 create(data) {
67 const now = new Date().toISOString();
68 const issue = {
69 id: this.generateId(),
70 reporterType: data.reporterType, // "Student" | "Indigene" | "Other"
71 anonymous: !!data.anonymous,
72 name: data.anonymous ? "" : data.name.trim(),
73 contact: data.anonymous ? "" : data.contact.trim(),
74 matricNo: data.matricNo ? data.matricNo.trim() : "",
75 community: data.community || "",
76 category: data.category,
77 severity: data.severity,
78 location: data.location.trim(),
79 incidentDate: data.incidentDate,
80 title: data.title.trim(),
81 description: data.description.trim(),
82 status: "Submitted",
83 createdAt: now,
84 history: [{ status: "Submitted", note: "Report received by IRO.", at: now }]
85 };
86 const list = this.getAll();
87 list.push(issue);
88 this.saveAll(list);
89 return issue;
90 },
91
92 /** Find one issue by its ticket ID (case-insensitive). */
93 findById(id) {
94 if (!id) return null;
95 const key = id.trim().toUpperCase();
96 return this.getAll().find(i => i.id === key) || null;
97 },
98
99 /** Change an issue's status and log the change in its history. */
100 updateStatus(id, status, note) {
101 const list = this.getAll();
102 const issue = list.find(i => i.id === id);
103 if (!issue) return null;
104 issue.status = status;
105 issue.history.push({ status, note: note || "", at: new Date().toISOString() });
106 this.saveAll(list);
107 return issue;
108 },
109
110 /** Remove an issue permanently (admin only). */
111 remove(id) {
112 this.saveAll(this.getAll().filter(i => i.id !== id));
113 },
114
115 /** Counts used by the admin dashboard. */
116 stats() {
117 const list = this.getAll();
118 const byStatus = {};
119 IRO.STATUSES.forEach(s => (byStatus[s] = 0));
120 list.forEach(i => byStatus[i.status]++);
121 return {
122 total: list.length,
123 open: list.length - byStatus["Resolved"] - byStatus["Closed"],
124 emergency: list.filter(i => i.severity === "Emergency" && i.status !== "Closed").length,
125 byStatus
126 };
127 }

132/** Escape text before inserting it into HTML (prevents script injection). */
133function escapeHtml(str) {
134 return String(str).replace(/[&<>"']/g, c => ({
135 "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;"
136 })[c]);
137}

139/** Format an ISO date for display, e.g. 7 Jul 2026, 14:05. */
140function formatDate(iso) {
141 return new Date(iso).toLocaleString("en-GB", {
142 day: "numeric", month: "short", year: "numeric", hour: "2-digit", minute: "2-digit"
143 });
144}

18 this.el("ticketInput").value = id;
19 this.show(id);

21 },
22
23 show(id) {
24 const issue = IssueStore.findById(id);
25 this.el("resultCard").classList.toggle("hidden", !issue);
26 this.el("trackError").textContent = issue ? "" : "No report found with that ticket ID. Check it and try again.";
27 if (!issue) return;
28
29 this.el("resTitle").textContent = issue.title;
30 this.el("resMeta").textContent =
31 `${issue.id} · ${issue.category} · ${issue.severity} severity · reported ${formatDate(issue.createdAt)}`;
32 this.el("resStatus").innerHTML = statusBadge(issue.status);
33 this.el("resHistory").innerHTML = issue.history.map(h => `
34 <li>
35 ${statusBadge(h.status)}
36 <small>${formatDate(h.at)}</small>
37 ${h.note ? `<div>${escapeHtml(h.note)}</div>` : ""}
38 </li>`).join("");
